Card-style layout showing all task fields. Implementation:

```zsh
# td-show - display full task details
# Autoloaded: file body = function body

local id_prefix="$1"

[[ -z "$id_prefix" ]] && {
  _td_error "Usage: todo show <id>"
  return 1
}

_td_storage_init

local tasks_file
tasks_file="$(_td_storage_list_path)"

local task_id
task_id="$(_td_resolve_task_id "$tasks_file" "$id_prefix")" || return 1

# Single JXA call to get all fields tab-separated
local output
output=$(TASK_ID="$task_id" JSON_DATA="$(cat "$tasks_file")" osascript -l JavaScript -e '
  function run() {
    var env = $.NSProcessInfo.processInfo.environment;
    var data = JSON.parse(env.objectForKey("JSON_DATA").js);
    var tid = env.objectForKey("TASK_ID").js;
    var t = null;
    for (var i = 0; i < data.tasks.length; i++) {
      if (data.tasks[i].id === tid) { t = data.tasks[i]; break; }
    }
    if (!t) return "ERROR:not_found";
    var tags = t.tags && t.tags.length > 0 ? t.tags.join(", ") : "";
    var ref = t.ref || "";
    var created = t.created || "";
    var updated = t.updated || "";
    // Return all fields tab-separated
    return [t.id, t.text, t.status, t.priority, tags, ref, created, updated].join("\t");
  }
' 2>/dev/null) || { _td_error "Failed to read task"; return 1; }

[[ "$output" == "ERROR:not_found" ]] && { _td_error "Task not found"; return 1; }

# Parse fields
local task_full_id task_text task_status task_priority task_tags task_ref task_created task_updated
IFS=$'\t' read -r task_full_id task_text task_status task_priority task_tags task_ref task_created task_updated <<< "$output"

# Relative time helper (inline JXA)
_td_show_relative_time() {
  local iso="$1"
  [[ -z "$iso" ]] && { printf "—"; return; }
  ISO_TIME="$iso" osascript -l JavaScript -e '
    function run() {
      var iso = $.NSProcessInfo.processInfo.environment.objectForKey("ISO_TIME").js;
      var d = new Date(iso);
      var now = new Date();
      var diff = Math.floor((now - d) / 1000);
      if (diff < 60) return diff + "s ago";
      if (diff < 3600) return Math.floor(diff / 60) + "m ago";
      if (diff < 86400) return Math.floor(diff / 3600) + "h ago";
      if (diff < 2592000) return Math.floor(diff / 86400) + "d ago";
      return Math.floor(diff / 2592000) + "mo ago";
    }
  ' 2>/dev/null
}

# Format output — card-style with aligned labels
local label_w=12
local dim="${TD_COLORS[dim]}"
local reset="${TD_COLORS[reset]}"
local bold="${TD_COLORS[bold]}"

printf "\n"
printf "  ${dim}%-${label_w}s${reset} %s\n" "ID" "${task_full_id}"
printf "  ${dim}%-${label_w}s${reset} %s\n" "Text" "${bold}${task_text}${reset}"
printf "  ${dim}%-${label_w}s${reset} ${TD_COLORS[cyan]}%s${reset}\n" "Status" "${task_status}"

# Priority with color
local pri_color="${TD_PRIORITY_COLORS[$task_priority]}"
printf "  ${dim}%-${label_w}s${reset} ${pri_color}%s${reset}\n" "Priority" "${task_priority}"

# Tags
if [[ -n "$task_tags" ]]; then
  printf "  ${dim}%-${label_w}s${reset} %s\n" "Tags" "${task_tags}"
else
  printf "  ${dim}%-${label_w}s${reset} ${dim}—${reset}\n" "Tags"
fi

# Ref
if [[ -n "$task_ref" ]]; then
  printf "  ${dim}%-${label_w}s${reset} ${TD_COLORS[blue]}%s${reset}\n" "Ref" "${task_ref}"
else
  printf "  ${dim}%-${label_w}s${reset} ${dim}—${reset}\n" "Ref"
fi

# Timestamps — relative with absolute in parentheses
local rel_created rel_updated
rel_created="$(_td_show_relative_time "$task_created")"
rel_updated="$(_td_show_relative_time "$task_updated")"
printf "  ${dim}%-${label_w}s${reset} %s ${dim}(%s)${reset}\n" "Created" "$rel_created" "$task_created"
printf "  ${dim}%-${label_w}s${reset} %s ${dim}(%s)${reset}\n" "Updated" "$rel_updated" "$task_updated"
printf "\n"
```

**4. Create `commands/td-ref` — attach/view ref URL (DATA-02):**

```zsh
# td-ref - attach or view reference URL for a task
# Autoloaded: file body = function body

local id_prefix="$1"
local ref_url="${2:-}"

[[ -z "$id_prefix" ]] && {
  _td_error "Usage: todo ref <id> [url]"
  return 1
}

_td_storage_init

local tasks_file
tasks_file="$(_td_storage_list_path)"

local task_id
task_id="$(_td_resolve_task_id "$tasks_file" "$id_prefix")" || return 1

if [[ -z "$ref_url" ]]; then
  # View mode: show current ref
  local current_ref
  current_ref=$(TASK_ID="$task_id" JSON_DATA="$(cat "$tasks_file")" osascript -l JavaScript -e '
    function run() {
      var env = $.NSProcessInfo.processInfo.environment;
      var data = JSON.parse(env.objectForKey("JSON_DATA").js);
      var tid = env.objectForKey("TASK_ID").js;
      for (var i = 0; i < data.tasks.length; i++) {
        if (data.tasks[i].id === tid) return data.tasks[i].ref || "";
      }
      return "";
    }
  ' 2>/dev/null)

  if [[ -z "$current_ref" ]]; then
    _td_success "No ref set for task ${task_id:0:8}"
  else
    printf "%s\n" "$current_ref"
  fi
else
  # Set mode: attach ref URL
  export TASK_ID="$task_id"
  export NEW_REF="$ref_url"
  _td_storage_modify "$tasks_file" '
    var tid = env.objectForKey("TASK_ID").js;
    var newRef = env.objectForKey("NEW_REF").js;
    var idx = data.tasks.findIndex(function(t) { return t.id === tid; });
    if (idx === -1) throw "Task not found";
    data.tasks[idx].ref = newRef;
    data.tasks[idx].updated = new Date().toISOString();
  ' || { unset TASK_ID NEW_REF; return 1; }
  unset TASK_ID NEW_REF
  _td_success "Ref set for task ${task_id:0:8}"
fi
```
